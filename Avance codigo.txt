function mapRow(headers, row) {

  const obj = {};

  headers.forEach((h, i) => {
    obj[h] = row[i];
  });

  return obj;
}

function login(nombre) {

  const user = DB_USERS_findByName(nombre);

  if (!user) {
    return {
      ok: false,
      message: "Usuario no encontrado"
    };
  }

  if (user.Activo !== true && user.Activo !== "TRUE") {
    return {
      ok: false,
      message: "Usuario inactivo"
    };
  }

  const session = {

    id: user.ID,

    nombre: user.Nombre,

    rol: user.Rol_principal,

    roles: user.Roles_permitidos
  };

  PropertiesService
    .getUserProperties()
    .setProperty("SESSION", JSON.stringify(session));

  return {
    ok: true,
    user: session
  };
}

function getSession() {

  const raw = PropertiesService
    .getUserProperties()
    .getProperty("SESSION");

  if (!raw) {
    return null;
  }

  return JSON.parse(raw);
}

function logout() {

  PropertiesService
    .getUserProperties()
    .deleteProperty("SESSION");

  return true;
}

05_Drive.gs
function uploadImage(base64, fileName){

  if(!base64){
    return "";
  }

  const folder =
    DriveApp.getFolderById(
      CONFIG.DRIVE_OBSERVACIONES
    );

  const bytes = Utilities.base64Decode(
    base64.split(",")[1]
  );

  const blob = Utilities.newBlob(
    bytes,
    "image/jpeg",
    fileName + ".jpg"
  );

  const file = folder.createFile(blob);

  file.setSharing(
    DriveApp.Access.ANYONE_WITH_LINK,
    DriveApp.Permission.VIEW
  );

  return file.getUrl();
}

function generarPDF(ids){

  // =========================
  // HTML
  // =========================
    
  const html =
    generarPreviewInforme(
      ids,
      "pdf"
  );

  // =========================
  // PDF
  // =========================

  const blob =

    HtmlService

      .createHtmlOutput(html)

      .getBlob()

      .getAs('application/pdf');

  // =========================
  // USER
  // =========================

  const user =
    getSession();

  // =========================
  // FECHA
  // =========================

  const fecha =

    Utilities.formatDate(

      new Date(),

      Session.getScriptTimeZone(),

      "yyyyMMdd_HHmmss"
    );

  // =========================
  // NOMBRE
  // =========================

  const nombre =

    `${user.nombre}-${user.rol}-${fecha}.pdf`;

  // =========================
  // BASE64
  // =========================

  const base64 =

    Utilities.base64Encode(
      blob.getBytes()
    );

  // =========================
  // RETURN
  // =========================

  return {

    base64:
      base64,

    nombre:
      nombre
  };
}

              Utilities.base64Encode(
                blob.getBytes()
              );

        Logger.log(err);

      // RETURN
      // =========================

      return {

        ID_OBSERVACION:
          obs.ID_OBSERVACION,

          obs.PASOS_REPRODUCIR,

          imageBase64
      };
    });

function generarPreviewInforme(
  ids,
  modo = "preview"
){

  // =========================
  // OBTENER OBSERVACIONES
  // =========================

  const observaciones =
    DB_OBS_getAll()

    .filter(x =>

      ids.includes(
        x.ID_OBSERVACION
      )
    )

    .map(obs => {

      let imageUrl = "";

      if(obs.CAPTURAS){

        const match =
          obs.CAPTURAS.match(
            /[-\w]{25,}/
          );

        if(match){

          const fileId =
            match[0];

          // =========================
          // PREVIEW HTML
          // =========================

          if(modo === "preview"){

            imageUrl =

              `https://lh3.googleusercontent.com/d/${fileId}`;
          }

          // =========================
          // PDF
          // =========================

          if(modo === "pdf"){

            try{

              const file =
                DriveApp.getFileById(
                  fileId
                );

              const blob =
                file

                  .getBlob()

                  .getAs(
                    MimeType.JPEG
                  );

              const base64 =

                Utilities.base64Encode(
                  blob.getBytes()
                );

              imageUrl =

                `data:image/jpeg;base64,${base64}`;

            }catch(err){

              Logger.log(err);
            }
          }
        }
      }
      // =========================
      // RETURN
      // =========================

      return {

        ID_OBSERVACION:
          obs.ID_OBSERVACION,

        QA:
          obs.QA,

        FECHA:
          Utilities.formatDate(
            new Date(obs.FECHA),
            Session.getScriptTimeZone(),
            "dd/MM/yyyy HH:mm"
          ),

        MODULO:
          obs.MODULO,

        DESCRIPCION_CORTA:
          obs.DESCRIPCION_CORTA,

        PASOS_REPRODUCIR:
          obs.PASOS_REPRODUCIR,

        //IMAGEN_BASE64:
        //  imagenBase64
        IMAGEN_URL:
          imageUrl
      };
    });

  // =========================
  // TEMPLATE
  // =========================

  const template =
    HtmlService.createTemplateFromFile(
      "template_informe"
    );

  template.fecha =
    Utilities.formatDate(
      new Date(),
      Session.getScriptTimeZone(),
      "dd/MM/yyyy HH:mm"
    );

  template.observaciones =
    observaciones;

  // =========================
  // HTML FINAL
  // =========================




  return template
    .evaluate()
    .getContent();
}

function DB_USERS_findByName(nombre) {

  const sh = SpreadsheetApp
    .getActive()
    .getSheetByName("USUARIOS");

  const data = sh.getDataRange().getValues();

  const headers = data[0];

  for (let i = 1; i < data.length; i++) {

    const row = data[i];

    const obj = mapRow(headers, row);

    if (
      obj.Nombre.toString().trim().toUpperCase() ===
      nombre.toString().trim().toUpperCase()
    ) {
      return obj;
    }
  }

  return null;
}

function dbGetTareaById(id){

  const sh =
    SpreadsheetApp
      .openById(CONFIG.ID_SHEET)
      .getSheetByName("TAREAS");

  const data =
    sh.getDataRange().getValues();

  const headers =
    data[0];

  for(let i = 1; i < data.length; i++){

    if(
      String(data[i][0]) === String(id)
    ){

      let obj = {};

      headers.forEach((h,j)=>{

        const value =
          data[i][j];

        obj[h] =

          value instanceof Date

          ?

          value.toISOString()

          :

          value;

      });

      return obj;
    }
  }

  return null;
}


function DB_TASK_update(
  payload
){

  const sh =
    SpreadsheetApp
      .getActive()
      .getSheetByName(
        "TAREAS"
      );

  const data =
    sh.getDataRange()
      .getValues();

  const headers =
    data[0];

  const col = {};

  headers.forEach((h,i) => {

    col[
      String(h)
        .trim()
        .toUpperCase()
    ] = i + 1;

  });

  for(
    let i = 1;
    i < data.length;
    i++
  ){

    if(

      String(
        data[i][
          col["ID_OBSERVACION"] - 1
        ]
      )

      ===

      String(
        payload.ID_OBSERVACION
      )

    ){

      sh.getRange(
        i + 1,
        col["MODULO"]
      ).setValue(
        payload.MODULO
      );

      sh.getRange(
        i + 1,
        col["DESCRIPCION_CORTA"]
      ).setValue(
        payload.DESCRIPCION_CORTA
      );

      sh.getRange(
        i + 1,
        col["PASOS_REPRODUCIR"]
      ).setValue(
        payload.PASOS_REPRODUCIR
      );

      sh.getRange(
        i + 1,
        col["PRIORIDAD"]
      ).setValue(
        payload.PRIORIDAD
      );

      sh.getRange(
        i + 1,
        col["USUARIOS_ASIGNADOS"]
      ).setValue(
        payload.USUARIOS_ASIGNADOS
      );


      // =========================
      // CAMBIAR A EN_PROCESO
      // SI EXISTE ALGUN ASIGNADO
      // =========================

      let tieneAsignados = false;

      try{

        const asignados =

          JSON.parse(
            payload.USUARIOS_ASIGNADOS || "{}"
          );

        tieneAsignados =

          Object.values(asignados)

            .some(lista =>

              Array.isArray(lista)

              &&

              lista.length > 0

            );

      }catch(error){

        Logger.log(error);

      }

      if(tieneAsignados){

        sh.getRange(
          i + 1,
          col["ESTADO"]
        ).setValue(
          "EN_PROCESO"
        );

      }

      // =========================
      // FECHA ASIGNACION
      // SOLO LA PRIMERA VEZ
      // =========================

      const fechaAsignacionActual =

        data[i][
          col["FECHA_ASIGNACION"] - 1
        ];

      if(

        !fechaAsignacionActual ||

        !(fechaAsignacionActual instanceof Date)

      ){

        sh.getRange(
          i + 1,
          col["FECHA_ASIGNACION"]
        ).setValue(
          new Date()
        );

      }
      
      return {
        ok:true
      };

    }

  }

  throw new Error(
    "No se encontró la tarea: " +
    payload.ID_OBSERVACION
  );

}

function TASK_SERVICE_list(
  user,
  filters = {}
){

  let data =
    DB_TASK_getAll();

  // =========================
  // FILTRO DE VISTA
  // =========================

  switch(filters.view){

    case "ELIMINADAS":

      data =
        data.filter(
          task => isDeleted(task)
        );

      break;

    default:

      data =
        data.filter(
          task => !isDeleted(task)
        );
  }

  // =========================
  // ROLES
  // =========================

  const roles =
    String(user.roles || "")
      .toUpperCase();

  // =========================
  // PM Y LIDERES VEN TODO
  // =========================

  if(

    roles.includes("PM") ||
    roles.includes("LIDER_QA") ||
    roles.includes("LIDER_DEV")

  ){

    return data;
  }
if(

  user.rol === "QA"

  ||

  user.rol === "DEV"

){

  return data.filter(task =>

    usuarioAsignadoATarea(

      task,

      user.nombre,

      user.rol

    )

  );

}
}


//helper para las tareas asignadas

function usuarioAsignadoATarea(
  tarea,
  nombre,
  rol
){

  try{

    const asignados =

      JSON.parse(
        tarea.USUARIOS_ASIGNADOS || "{}"
      );

    const lista =

      asignados[rol] || [];

    return lista.some(x =>

      String(x)
        .trim()
        .toUpperCase()

      ===

      String(nombre)
        .trim()
        .toUpperCase()

    );

  }
  catch(error){

    Logger.log(error);

    return false;

  }

}

function isDeleted(obs){

  return String(obs.ELIMINADO || "")
    .trim()
    .toUpperCase() === "TRUE";
}

function OBS_SERVICE_list(
  user,
  filters = {}
) {

  let data =
    DB_OBS_getAll();

  // =========================
  // FILTRO DE VISTA
  // =========================

  switch(filters.view){

    case "ELIMINADAS":

      data =
        data.filter(
          obs => isDeleted(obs)
        );

      break;

    default:

      data =
        data.filter(
          obs => !isDeleted(obs)
        );
  }

  // =========================
  // ROLES
  // =========================

  const roles =
    String(user.roles || "")
      .toUpperCase();

  // =========================
  // PM Y LIDERES VEN TODO
  // =========================

  if(

    roles.includes("PM") ||
    roles.includes("LIDER_QA") ||
    roles.includes("LIDER_DEV")

  ){

    return data;
  }

  // =========================
  // QA VE SOLO SUS OBS
  // =========================

  if(user.rol === "QA"){

    return data.filter(x =>

      String(x.QA)
        .trim()
        .toUpperCase()

      ===

      String(user.nombre)
        .trim()
        .toUpperCase()
    );
  }

  // =========================
  // DEV VE SOLO ASIGNADAS
  // =========================

  if(user.rol === "DEV"){

    return data.filter(x =>

      String(x.DEV_ASIGNADO)
        .trim()
        .toUpperCase()

      ===

      String(user.nombre)
        .trim()
        .toUpperCase()
    );
  }

  // =========================
  // DEFAULT
  // =========================

  return [];
}

    SpreadsheetApp

      .getValues();

    headers.indexOf(

    );

  const moduloCol =
    headers.indexOf(
      "MODULO"
    );

  const descCol =
    headers.indexOf(
      "DESCRIPCION_CORTA"
    );

  const pasosCol =
    headers.indexOf(
      "PASOS_REPRODUCIR"
    );

  const prioridadCol =
    headers.indexOf(
      "PRIORIDAD"
    );

  const capturasCol =
    headers.indexOf(
      "CAPTURAS"
    );

  for(

    let i = 1;

    i < data.length;

    i++

  ){

    if(

      String(
        data[i][idCol]
      )

      ===

      String(
        payload
          .ID_OBSERVACION
      )

    ){

      // =====================
      // TEXTO
      // =====================

      data[i][moduloCol] =

        payload.MODULO;

      data[i][descCol] =

        payload
          .DESCRIPCION_CORTA;

      data[i][pasosCol] =

        payload
          .PASOS_REPRODUCIR;

      data[i][prioridadCol] =

        payload.PRIORIDAD;

      // =====================
      // IMAGEN
      // =====================

      const tieneImagen =

        data[i][capturasCol];

      if(

        !tieneImagen &&

        payload.Imagen

      ){

        Logger.log(
          "Subiendo imagen..."
        );

        const imageUrl =

          uploadImage(

            payload.Imagen,

            payload
              .ID_OBSERVACION

          );

        Logger.log(
          imageUrl
        );

        data[i][capturasCol] =
          imageUrl;
      }

      break;
    }
  }

  sheet
    .getDataRange()
    .setValues(data);

  return true;

<script>

function descargarPDF(ids){

  showLoading(
    "Generando PDF..."
  );

  google.script.run

    .withSuccessHandler(res => {

      hideLoading();

      // =========================
      // BASE64 -> BLOB
      // =========================

      const bytes =

        atob(res.base64);

      const array =

        new Uint8Array(
          bytes.length
        );

      for(

        let i = 0;

        i < bytes.length;

        i++

      ){

        array[i] =
          bytes.charCodeAt(i);
      }

      // =========================
      // PDF BLOB
      // =========================

      const blob =

        new Blob(

          [array],

          {
            type:'application/pdf'
          }
        );

      // =========================
      // URL
      // =========================

      const url =

        URL.createObjectURL(
          blob
        );

      // =========================
      // DOWNLOAD
      // =========================

      const a =
        document.createElement(
          'a'
        );

      a.href =
        url;

      a.download =
        res.nombre;

      document.body
        .appendChild(a);

      a.click();

      a.remove();

      URL.revokeObjectURL(
        url
      );

      // =========================
      // SUCCESS
      // =========================

      Swal.fire({

        icon:'success',

        title:'PDF generado'
      });

    })

    .withFailureHandler(err => {

      hideLoading();

      console.error(err);

      Swal.fire({

        icon:'error',

        title:'Error PDF',

        text:err.message
      });

    })

    .generarPDF(ids);
}

</script>

function doLogin() {

  const nombre =
    document.getElementById("txtUser").value;
  
  showLoading(
    "Ingresando..."
  );

  google.script.run
    .withSuccessHandler(res => {
      
      hideLoading();
      
      if (!res.ok) {

        hideLoading();

        Swal.fire({
          icon: 'error',
          title: 'Error',
          text: res.message
        });

        return;
      }

      window.currentUser = res.user;

      window.currentRole =   
        res.user.rol;

      CURRENT_USER = res.user;


      document
        .getElementById("sidebar")
        .classList.remove("d-none");

      loadDashboard();

      document
        .getElementById("navbarUser")
        .classList.remove("d-none");

      document
        .getElementById("txtNavbarUser")
        .innerText = res.user.nombre;

      cargarRolesNavbar(res.user);

    })
    .login(nombre);
}

let IMAGE_BASE64 = "";

/* =========================
   PREVIEW + COMPRESION
========================= */

document.addEventListener(
  "change",
  async function(e){

    if(e.target.id !== "obsImagen"){
      return;
    }

    const file = e.target.files[0];

    if(!file){
      return;
    }

    const compressed =
      await compressImage(file);

    IMAGE_BASE64 = compressed;

    document
      .getElementById("previewImagen")
      .src = compressed;

    document
      .getElementById("previewImagen")
      .style.display = "block";
  }
);

/* =========================
   COMPRESS IMAGE
========================= */

function compressImage(file){

  return new Promise((resolve)=>{

    const reader = new FileReader();

    reader.readAsDataURL(file);

    reader.onload = function(event){

      const img = new Image();

      img.src = event.target.result;

      img.onload = function(){

        const canvas =
          document.createElement("canvas");

        const MAX_WIDTH = 1200;

        let width = img.width;
        let height = img.height;

        if(width > MAX_WIDTH){

          height *= MAX_WIDTH / width;
          width = MAX_WIDTH;
        }

        canvas.width = width;
        canvas.height = height;

        const ctx =
          canvas.getContext("2d");

        ctx.drawImage(
          img,
          0,
          0,
          width,
          height
        );

        // calidad 0.5 = muy liviano
        const compressed =
          canvas.toDataURL(
            "image/jpeg",
            0.5
          );

        resolve(compressed);
      };
    };
  });
}

<script>
function dividirTexto(

  texto,
  maxChars = 65

){

  if(!texto){
    return [];
  }

  // =========================
  // NORMALIZAR
  // =========================

  texto =

    String(texto)
      .replace(/\r/g,'');

  // =========================
  // SPLIT SALTOS REALES
  // =========================

  const bloques =

    texto.split('\n');

  const resultado = [];

  // =========================
  // RECORRER BLOQUES
  // =========================

  for(

    const bloque of bloques

  ){

    // =========================
    // LINEA VACIA
    // =========================

    if(!bloque.trim()){

      resultado.push('');
      continue;
    }

    // =========================
    // WRAP NORMAL
    // =========================

    const palabras =

      bloque.split(' ');

    let linea = '';

    for(

      const palabra of palabras

    ){

      if(

        (linea + palabra).length
        > maxChars

      ){

        resultado.push(
          linea.trim()
        );

        linea =
          palabra + ' ';

      }else{

        linea +=
          palabra + ' ';
      }
    }

    // =========================
    // PUSH FINAL
    // =========================

    if(linea.trim()){

      resultado.push(
        linea.trim()
      );
    }
  }

  return resultado;
}
</script>

          page.drawText(

            titulo,

            {

              x:
                (595 / 2)
                - (tituloWidth / 2),

              y:y,

              size:16
            }
          );

          y -= 25;
          page.drawText(

            `Usuario: ${data[0]?.QA || ""}`,

            {
              x:40,
              y:y,
              size:14
            }
          );

          y -= 15;

          page.drawLine({

            start:{
              x:40,
              y:y
            },

            end:{
              x:550,
              y:y
            },

            thickness:1
          });

          y -= 25;
          
          for(           

            const obs of data

          ){
            // =========================
            // BLOQUE VISUAL
            // =========================

            // =========================
            // NUEVA PAGINA
            // =========================

            if(y < 250){

              page =

                pdfDoc.addPage([
                  595,
                  842
                ]);

              y = 780;
            }

            // =========================
            // TITULO
            // =========================

            page.drawText(

              `OBSERVACION: ${obs.ID_OBSERVACION}`,

              {
                x:40,
                y:y,
                size:14
              }
            );

            y -= 25;

            // =========================
            // IMAGEN
            // =========================

            if(obs.IMAGEN_BASE64){

              try{

                const image =

                  await pdfDoc
                    .embedJpg(
                      obs.IMAGEN_BASE64
                    );

                // =========================
                // CONTENEDOR
                // =========================

                const maxWidth  = 120;
                const maxHeight = 180;

                // =========================
                // DIMENSIONES ORIGINALES
                // =========================

                const originalWidth  = image.width;
                const originalHeight = image.height;

                // =========================
                // ESCALA
                // =========================

                const scale = Math.min(

                  maxWidth  / originalWidth,

                  maxHeight / originalHeight
                );

                // =========================
                // TAMAÑO FINAL
                // =========================

                const finalWidth  =
                  originalWidth * scale;

                const finalHeight =
                  originalHeight * scale;

                // =========================
                // CENTRADO VERTICAL
                // =========================

                const offsetY =
                  (maxHeight - finalHeight) / 2;

                // =========================
                // DIBUJAR CONTENEDOR
                // =========================

                // =========================
                // IMAGEN
                // =========================

                page.drawImage(

                  image,

                  {

                    x:40,

                    y:
                      y - 210 + offsetY,

                    width:finalWidth,

                    height:finalHeight
                  }
                );

              }catch(err){

                console.error(err);
              }
            }
            // =========================
            // POSICION TEXTO
            // =========================

            const textX = 170;

            let textY = y;

            // =========================
            // MODULO
            // =========================

            page.drawText(

              `Modulo: ${obs.MODULO || ""}`,

              {
                x:textX,
                y:textY,
                size:12
              }
            );

            textY -= 25;

            // =========================
            // ESTADO
            // =========================

            page.drawText(

              `Estado: ${obs.ESTADO || ""}`,

              {
                x:textX,
                y:textY,
                size:12
              }
            );

            textY -= 25;

            // =========================
            // PRIORIDAD
            // =========================

            page.drawText(

              `Prioridad: ${obs.PRIORIDAD || ""}`,

              {
                x:textX,
                y:textY,
                size:12
              }
            );

            textY -= 35;

            // =========================
            // DESCRIPCION
            // =========================

            page.drawText(

              `Descripcion:`,

              {
                x:textX,
                y:textY,
                size:12
              }
            );

            textY -= 20;

            // =========================
            // TEXTO WRAP
            // =========================

            const descripcionLineas =

              dividirTexto(

                obs.DESCRIPCION_CORTA,

                90
              );

            for(

              const linea of descripcionLineas

            ){

              page.drawText(

                linea,

                {
                  x:textX,
                  y:textY,
                  size:10
                }
              );

              textY -= 14;
            }

            // =========================
            // ESPACIO
            // =========================

            textY -= 15;

            // =========================
            // PASOS
            // =========================

            page.drawText(

              `Pasos:`,

              {
                x:textX,
                y:textY,
                size:12
              }
            );

            textY -= 20;

            // =========================
            // PASOS TEXTO
            // =========================

            const pasosLineas =

              dividirTexto(

                obs.PASOS_REPRODUCIR,

                90
              );

            for(

              const linea of pasosLineas

            ){

              page.drawText(

                linea,

                {
                  x:textX,
                  y:textY,
                  size:10
                }
              );

              textY -= 14;              
            }

            // =========================
            // NUEVA POSICION GLOBAL
            // =========================

            y = Math.min(

              y - 240,

              textY - 40
            );
          }
          // =========================
          // SAVE PDF
          // =========================

          const pdfBytes =

            await pdfDoc.save();

          // =========================
          // BLOB
          // =========================

          const blob =

            new Blob(

              [pdfBytes],

              {
                type:'application/pdf'
              }
            );

          // =========================
          // DOWNLOAD
          // =========================

          const url =

            URL.createObjectURL(
              blob
            );

          const a =

            document.createElement(
              'a'
            );

          a.href =
            url;

          const usuario =

            document.getElementById(
              "txtNavbarUser"
            )

            .innerText

            .trim()

            .replace(/\s+/g,"_");

          const rol =

            document.getElementById(
              "cmbRoles"
            )
            .value;

          const fecha =

            new Date()

              .toISOString()

              .replace(/[:.]/g,"-");

          a.download =

            `${usuario}-${rol}-${fecha}.pdf`;

          document.body
            .appendChild(a);

          a.click();

          a.remove();

          URL.revokeObjectURL(
            url
          );

          hideLoading();

        }catch(err){

          hideLoading();

          console.error(err);

          Swal.fire({

            icon:'error',

            title:'PDF Error',

            text:err.message
          });
        }
      }
    )

    .withFailureHandler(err => {

      hideLoading();

      console.error(err);

      Swal.fire({

        icon:'error',

        title:'Backend Error',

        text:err.message
      });

    })

    .obtenerDataPDF(ids);
}

<script>
async function enviarComentario(){

  const mensaje =
    document
      .getElementById(
        "txtNuevoComentario"
      )
      .value
      .trim();

  if(!mensaje){

    Swal.fire({

      icon:"warning",

      title:"Mensaje vacío",

      text:
        "Debe escribir un comentario."

    });

    return;
  }

  const obs =
    window.obsChatActual;

  const usuario =
    document
      .getElementById(
        "txtNavbarUser"
      )
      .innerText
      .trim();

  const rol =
    document
      .getElementById(
        "cmbRoles"
      )
      .value;

  const payload = {

    id:
      obs.ID_OBSERVACION,

    usuario,

    rol,

    mensaje,

    marcarRevision:false,

    aprobarCorreccion:false,

    rechazarCorreccion:false

  };

  try{

    // =====================
    // DEV ASIGNADO
    // =====================

    const esDevAsignado =

      rol === "DEV"

      &&

      usuario ===
      obs.DEV_ASIGNADO;

    if(esDevAsignado){

      const result =
        await Swal.fire({

          title:
            "¿Marcar como listo para revisión?",

          text:
            "La observación cambiará a EN_REVISION.",

          icon:"question",

          showCancelButton:true,

          confirmButtonText:"Sí",

          cancelButtonText:"No"

        });

      payload.marcarRevision =
        result.isConfirmed;
    }

    // =====================
    // QA Y LIDERES
    // =====================

    const puedeValidar =

      [

        "QA",
        "LIDER_QA",
        "LIDER_DEV"

      ].includes(
        rol
      );

    if(

      puedeValidar

      &&

      obs.ESTADO ===
      "EN_REVISION"

    ){

      const result =
        await Swal.fire({

          title:
            "Validación",

          text:
            "¿Aceptar o rechazar la corrección?",

          icon:"question",

          showDenyButton:true,

          showCancelButton:true,

          confirmButtonText:
            "Aceptar",

          denyButtonText:
            "Rechazar",

          cancelButtonText:
            "Solo comentar"

        });

      if(
        result.isConfirmed
      ){

        payload.aprobarCorreccion =
          true;
      }

      else if(
        result.isDenied
      ){

        payload.rechazarCorreccion =
          true;
      }
    }

    showLoading(
      "Guardando comentario..."
    );

    await runServer(

      "OBS_SERVICE_addComment",

      payload

    );

    document
      .getElementById(
        "txtNuevoComentario"
      )
      .value = "";

    abrirComentarios(
      obs.ID_OBSERVACION
    );

    Swal.fire({

      icon:"success",

      title:"Comentario guardado",

      timer:1500,

      showConfirmButton:false

    });

  }
  catch(error){

    console.error(error);

    Swal.fire({

      icon:"error",

      title:"Error",

      text:error.message

    });

  }
  finally{

    hideLoading();
  }
}
</script>
